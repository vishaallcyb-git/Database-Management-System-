CREATE DATABASE BankDB;
USE BankDB;

CREATE TABLE Transactions (
    TxnID BIGINT PRIMARY KEY,
    AccountNo VARCHAR(12),
    TxnTime DATETIME,
    Type VARCHAR(20),
    Amount DECIMAL(12,2),
    Branch VARCHAR(20),
    Channel VARCHAR(10),
    Narration VARCHAR(80),
    Balance DECIMAL(12,2)
);

-- Secondary index on AccountNo
CREATE INDEX idx_account
ON Transactions(AccountNo);

-- Secondary index on Branch
CREATE INDEX idx_branch
ON Transactions(Branch);

-- Index for transaction time
CREATE INDEX idx_time
ON Transactions(TxnTime);
