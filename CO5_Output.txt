CREATE DATABASE RiverGate;
USE RiverGate;

CREATE TABLE CanalPool (
    PoolID INT PRIMARY KEY,
    PoolName VARCHAR(30),
    Balance INT
);

CREATE TABLE Farm (
    FarmID INT PRIMARY KEY,
    FarmName VARCHAR(30),
    DailyLimit INT
);

CREATE TABLE Allocation (
    RequestID INT PRIMARY KEY,
    FarmID INT,
    SourcePool INT,
    DestinationPool INT,
    Amount INT,
    FOREIGN KEY (FarmID) REFERENCES Farm(FarmID)
);
