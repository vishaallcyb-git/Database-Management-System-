CREATE DATABASE SwiftSpan;
USE SwiftSpan;

CREATE TABLE Sender (
    SenderID INT PRIMARY KEY,
    SenderName VARCHAR(50),
    RouteGroup VARCHAR(30)
);

CREATE TABLE Service (
    ServiceID INT PRIMARY KEY,
    ServiceCode VARCHAR(20) UNIQUE,
    ServiceName VARCHAR(50),
    VolumeWeight DECIMAL(5,2)
);

CREATE TABLE Coordinator (
    CoordinatorID INT PRIMARY KEY,
    CoordinatorName VARCHAR(50),
    Hub VARCHAR(30)
);

CREATE TABLE Assignment (
    SenderID INT,
    ServiceID INT,
    CoordinatorID INT,
    PRIMARY KEY (SenderID, ServiceID),
    FOREIGN KEY (SenderID) REFERENCES Sender(SenderID),
    FOREIGN KEY (ServiceID) REFERENCES Service(ServiceID),
    FOREIGN KEY (CoordinatorID) REFERENCES Coordinator(CoordinatorID)
);

CREATE TABLE Observation (
    SenderID INT,
    ServiceID INT,
    DispatchWave INT,
    Completion INT,
    PRIMARY KEY (SenderID, ServiceID, DispatchWave),
    FOREIGN KEY (SenderID, ServiceID)
        REFERENCES Assignment(SenderID, ServiceID)
);

SELECT * FROM Observation
WHERE SenderID = 1 AND DispatchWave = 1;

-- Q2: Find observations with completion >= 39
SELECT * FROM Observation
WHERE Completion >= 39;

-- Q3: Find observations below 39
SELECT * FROM Observation
WHERE Completion < 39;

-- Q4: Average completion by coordinator
SELECT C.CoordinatorName, AVG(O.Completion) AS AverageCompletion
FROM Coordinator C
JOIN Assignment A ON C.CoordinatorID = A.CoordinatorID
JOIN Observation O
ON A.SenderID = O.SenderID AND A.ServiceID = O.ServiceID
GROUP BY C.CoordinatorID, C.CoordinatorName;

-- Q5: Senders with low completion in at least 2 waves
SELECT SenderID
FROM Observation
WHERE Completion < 39
GROUP BY SenderID
HAVING COUNT(DISTINCT DispatchWave) >= 2;
CREATE VIEW SenderTrackingView AS
SELECT S.SenderID, S.SenderName,
       O.ServiceID, O.DispatchWave, O.Completion
FROM Sender S
JOIN Observation O ON S.SenderID = O.SenderID;

CREATE VIEW CoordinatorWorkView AS
SELECT C.CoordinatorName, AVG(O.Completion) AS AverageCompletion
FROM Coordinator C
JOIN Assignment A ON C.CoordinatorID = A.CoordinatorID
JOIN Observation O
ON A.SenderID = O.SenderID AND A.ServiceID = O.ServiceID
GROUP BY C.CoordinatorName;
