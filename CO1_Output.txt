CREATE DATABASE IF NOT EXISTS meridian_museum;
USE meridian_museum;

-- 1. PARTY
CREATE TABLE Party (
    PartyID BIGINT AUTO_INCREMENT PRIMARY KEY,
    PartyType ENUM('PERSON', 'ORGANIZATION') NOT NULL
);

-- 2. PERSON
CREATE TABLE Person (
    PartyID BIGINT PRIMARY KEY,
    GivenName VARCHAR(80) NOT NULL,
    MiddleName VARCHAR(80),
    FamilyName VARCHAR(80) NOT NULL,
    Email VARCHAR(254),
    FOREIGN KEY (PartyID)
        REFERENCES Party(PartyID)
);

-- 3. ORGANIZATION
CREATE TABLE Organization (
    PartyID BIGINT PRIMARY KEY,
    LegalName VARCHAR(180) NOT NULL,
    RegistrationNo VARCHAR(80) UNIQUE,
    FOREIGN KEY (PartyID)
        REFERENCES Party(PartyID)
);

-- 4. EMPLOYEE
CREATE TABLE Employee (
    PartyID BIGINT PRIMARY KEY,
    EmployeeNo VARCHAR(30) UNIQUE NOT NULL,
    Role ENUM('CURATOR', 'REGISTRAR', 'OTHER') NOT NULL,
    HireDate DATE NOT NULL,
    FOREIGN KEY (PartyID)
        REFERENCES Person(PartyID)
);

-- 5. LENDER ROLE
CREATE TABLE LenderRole (
    PartyID BIGINT PRIMARY KEY,
    ActiveFlag BOOLEAN DEFAULT TRUE,
    FOREIGN KEY (PartyID)
        REFERENCES Party(PartyID)
);

-- 6. PARTY PHONE
CREATE TABLE PartyPhone (
    PartyID BIGINT NOT NULL,
    PhoneNo VARCHAR(30) NOT NULL,
    Label VARCHAR(20),
    PRIMARY KEY (PartyID, PhoneNo),
    FOREIGN KEY (PartyID)
        REFERENCES Party(PartyID)
);

-- 7. COLLECTION UNIT
CREATE TABLE CollectionUnit (
    CollectionUnitID INT AUTO_INCREMENT PRIMARY KEY,
    UnitName VARCHAR(100) UNIQUE NOT NULL
);

-- 8. ARTIFACT
CREATE TABLE Artifact (
    ArtifactID BIGINT AUTO_INCREMENT PRIMARY KEY,
    AccessionNo VARCHAR(40) UNIQUE NOT NULL,
    CollectionUnitID INT NOT NULL,
    Title VARCHAR(200) NOT NULL,
    CollectionType VARCHAR(60) NOT NULL,
    CurrentValue DECIMAL(12,2),
    FOREIGN KEY (CollectionUnitID)
        REFERENCES CollectionUnit(CollectionUnitID)
);

-- 9. LOAN
CREATE TABLE Loan (
    LoanID BIGINT AUTO_INCREMENT PRIMARY KEY,
    LenderPartyID BIGINT NOT NULL,
    StartDate DATE NOT NULL,
    DueDate DATE NOT NULL,
    ReturnDate DATE,
    Status ENUM(
        'PLANNED',
        'ACTIVE',
        'CLOSED',
        'CANCELLED'
    ) NOT NULL,

    FOREIGN KEY (LenderPartyID)
        REFERENCES LenderRole(PartyID),

    CHECK (DueDate >= StartDate),
    CHECK (
        ReturnDate IS NULL
        OR ReturnDate >= StartDate
    )
);

-- 10. LOAN OBJECT
CREATE TABLE LoanObject (
    LoanObjectID BIGINT AUTO_INCREMENT PRIMARY KEY,
    LoanID BIGINT NOT NULL,
    ArtifactID BIGINT NOT NULL,
    IntakeDate DATE NOT NULL,
    ReturnDate DATE,
    LineValue DECIMAL(12,2),
    Notes VARCHAR(500),

    UNIQUE (LoanID, ArtifactID),

    FOREIGN KEY (LoanID)
        REFERENCES Loan(LoanID),

    FOREIGN KEY (ArtifactID)
        REFERENCES Artifact(ArtifactID),

    CHECK (
        ReturnDate IS NULL
        OR ReturnDate >= IntakeDate
    )
);

-- 11. CONDITION REPORT
CREATE TABLE ConditionReport (
    ReportID BIGINT AUTO_INCREMENT PRIMARY KEY,
    LoanObjectID BIGINT NOT NULL,
    ReportDateTime DATETIME NOT NULL,

    ReportType ENUM(
        'INTAKE',
        'DURING_LOAN',
        'RETURN'
    ) NOT NULL,

    ConditionGrade VARCHAR(30),
    Observations TEXT NOT NULL,
    EvidenceURI VARCHAR(500),

    FOREIGN KEY (LoanObjectID)
        REFERENCES LoanObject(LoanObjectID)
);
-- 12. EXHIBITION
CREATE TABLE Exhibition (
    ExhibitionID BIGINT AUTO_INCREMENT PRIMARY KEY,
    ExhibitionName VARCHAR(180) NOT NULL,
    StartDate DATE NOT NULL,
    EndDate DATE NOT NULL,
    CHECK (EndDate >= StartDate)
);

-- 13. DISPLAY SLOT
CREATE TABLE DisplaySlot (
    ExhibitionID BIGINT NOT NULL,
    SlotNo INT NOT NULL,
    Zone VARCHAR(60),
    Capacity INT DEFAULT 1,

    PRIMARY KEY (ExhibitionID, SlotNo),

    FOREIGN KEY (ExhibitionID)
        REFERENCES Exhibition(ExhibitionID),

    CHECK (SlotNo > 0),
    CHECK (Capacity > 0)
);

-- 14. PLACEMENT
CREATE TABLE Placement (
    PlacementID BIGINT AUTO_INCREMENT PRIMARY KEY,
    LoanObjectID BIGINT NOT NULL,
    ExhibitionID BIGINT NOT NULL,
    SlotNo INT NOT NULL,
    FromDate DATE NOT NULL,
    ToDate DATE,

    FOREIGN KEY (LoanObjectID)
        REFERENCES LoanObject(LoanObjectID),

    FOREIGN KEY (ExhibitionID, SlotNo)
        REFERENCES DisplaySlot(ExhibitionID, SlotNo),

    CHECK (
        ToDate IS NULL OR ToDate >= FromDate
    )
);

-- 15. ARTIFACT OWNER
CREATE TABLE ArtifactOwner (
    ArtifactID BIGINT NOT NULL,
    PartyID BIGINT NOT NULL,
    ValidFrom DATE NOT NULL,
    ValidTo DATE,
    SharePercent DECIMAL(5,2) NOT NULL,

    PRIMARY KEY (ArtifactID, PartyID, ValidFrom),

    FOREIGN KEY (ArtifactID)
        REFERENCES Artifact(ArtifactID),

    FOREIGN KEY (PartyID)
        REFERENCES Party(PartyID),

    CHECK (SharePercent > 0 AND SharePercent <= 100),

    CHECK (
        ValidTo IS NULL OR ValidTo >= ValidFrom
    )
);

SELECT LoanID, LenderPartyID, DueDate
FROM Loan
WHERE ReturnDate IS NULL
AND DueDate < CURRENT_DATE;

SELECT
    A.ArtifactID,
    A.AccessionNo,
    A.Title
FROM Artifact A
JOIN LoanObject LO
    ON A.ArtifactID = LO.ArtifactID
LEFT JOIN ConditionReport CR
    ON LO.LoanObjectID = CR.LoanObjectID
WHERE CR.ReportID IS NULL;

SELECT
    A.AccessionNo,
    A.Title,
    L.LoanID,
    L.StartDate,
    L.DueDate,
    LO.ReturnDate
FROM Artifact A
JOIN LoanObject LO
    ON A.ArtifactID = LO.ArtifactID
JOIN Loan L
    ON LO.LoanID = L.LoanID
ORDER BY L.StartDate;

SELECT
    LO.LoanObjectID,
    LO.LoanID,
    LO.ArtifactID
FROM LoanObject LO
WHERE NOT EXISTS (
    SELECT 1
    FROM ConditionReport CR
    WHERE CR.LoanObjectID = LO.LoanObjectID
    AND CR.ReportType = 'INTAKE'
);
